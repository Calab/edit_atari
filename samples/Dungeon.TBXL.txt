400 PROC INIT_WORLD
410   FOR IDX=1 TO MAPW*MAPH
420     MAP$(IDX,IDX)=".":VIS$(IDX,IDX)="0"
430   NEXT IDX
440   FOR IDX=1 TO MAPW

460   NEXT IDX
470   FOR IDX=1 TO MAPH

500   NEXT IDX
510   FOR IDX=4 TO 16 STEP 3

570   FOR IDX=1 TO MAXMON

600     MXP(IDX)=15+IDX*5:MALIVE(IDX)=1
610   NEXT IDX
620   FOR IDX=1 TO MAXITM

680 PROC REVEAL_FOG
690   FOR JDX=-1 TO 1
700     FOR KDX=-1 TO 1
710       TX=PLRX+JDX:TY=PLRY+KDX

730         ADDR=(TY-1)*MAPW+TX
740         VIS$(ADDR,ADDR)="1"
750       ENDIF
760     NEXT KDX
770   NEXT JDX
780 ENDPROC
790 PROC READ_INPUT
800   CMD=0:CH=PEEK(764)
810   IF CH=255 THEN RETURN
820   POKE 764,255
830   IF CH=46 OR CH=14 THEN CMD=1 : REM W / UP
840   IF CH=62 OR CH=15 THEN CMD=2 : REM S / DOWN
850   IF CH=42 OR CH=13 THEN CMD=3 : REM A / LEFT
860   IF CH=45 OR CH=12 THEN CMD=4 : REM D / RIGHT
870   IF CH=37 THEN CMD=5 : REM R (REST/HEAL)
880   IF CH=28 THEN GAMEOVER=1 : REM ESC
890 ENDPROC
900 PROC HANDLE_MOVE
910   MOVEOK=0:LASTX=PLRX:LASTY=PLRY:TX=PLRX:TY=PLRY
920   IF CMD=1 THEN TY=PLRY-1
930   IF CMD=2 THEN TY=PLRY+1
940   IF CMD=3 THEN TX=PLRX-1
950   IF CMD=4 THEN TX=PLRX+1
960   IF CMD=5 THEN EXEC DO_REST:RETURN
970   IF TX<1 OR TX>MAPW OR TY<1 OR TY>MAPH THEN RETURN
980   ADDR=(TY-1)*MAPW+TX
990   SYM=ASC(MAP$(ADDR,ADDR))
1000  IF SYM=35 THEN MSGTXT$="A SOLID WALL BLOCKS YOU.":RETURN : REM #
1010  IF SYM=43 AND PLRKEYS<=0 THEN MSGTXT$="LOCKED DOOR. NEED A KEY.":RETURN : REM +
1020  IF SYM=43 AND PLRKEYS>0 THEN PLRKEYS=PLRKEYS-1:MAP$(ADDR,ADDR)=".":MSGTXT$="UNLOCKED THE DOOR."

1070    ENDIF
1080  NEXT IDX
1090  PLRX=TX:PLRY=TY:MOVEOK=1:TURNS=TURNS+1
1100  IF TORCH>0 THEN TORCH=TORCH-1

1120  EXEC CHECK_ITEMS
1130 ENDPROC

1210  ENDIF
1220  TURNS=TURNS+1:MOVEOK=1

1240 PROC CHECK_ITEMS
1250  FOR IDX=1 TO MAXITM
1260    IF IACTIVE(IDX)=1 AND IX(IDX)=PLRX AND IY(IDX)=PLRY THEN
1270      IACTIVE(IDX)=0
1280      IF ITYPE(IDX)=1 THEN PLRHP=PLRHP+IVAL(IDX):IF PLRHP>PLRMAXHP THEN PLRHP=PLRMAXHP:ENDIF:MSGTXT$="FOUND POTION (+HP)."
1290      IF ITYPE(IDX)=2 THEN PLRGOLD=PLRGOLD+IVAL(IDX):SCORE=SCORE+IVAL(IDX):MSGTXT$="FOUND GOLD COINS."
1300      IF ITYPE(IDX)=3 THEN PLRKEYS=PLRKEYS+1:MSGTXT$="FOUND AN IRON KEY."
1310    ENDIF

1330 ENDPROC

1350  HITROLL=RAND(10)+1
1360  DMG=PLRATK+HITROLL-MDEF(IDX)

1380  MHP(IDX)=MHP(IDX)-DMG
1390  IF MHP(IDX)<=0 THEN
1400    MALIVE(IDX)=0:PLRXP=PLRXP+MXP(IDX):SCORE=SCORE+MXP(IDX)*2

1430  ELSE
1440    MSGTXT$="HIT FOE FOR "

1490  IF PLRXP>=PLRNEXTXP THEN
1500    PLRLVL=PLRLVL+1:PLRXP=PLRXP-PLRNEXTXP:PLRNEXTXP=PLRNEXTXP+25
1510    PLRMAXHP=PLRMAXHP+15:PLRHP=PLRMAXHP:PLRATK=PLRATK+3:PLRDEF=PLRDEF+1
1520    MSGTXT$="LEVEL UP! STATS INCREASED."

1550 PROC STEP_MONSTERS
1560  FOR IDX=1 TO MAXMON
1570    IF MALIVE(IDX)=1 THEN
1580      IF ABS(MX(IDX)-PLRX)+ABS(MY(IDX)-PLRY)=1 THEN
1590        EVADE=RAND(20)+1

1730        IF MX(IDX)>PLRX THEN TX=MX(IDX)-1
1740        IF MY(IDX)<PLRY THEN TY=MY(IDX)+1
1750        IF MY(IDX)>PLRY THEN TY=MY(IDX)-1
